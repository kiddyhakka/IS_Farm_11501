步驟指南：如何建立免費 Google Sheets 後端 API

  1. 開啟全新的 Google 試算表，並建立三張工作表：FieldRecords, RawInbound, Shipping。
  2. 點選 「擴充功能」 ➔ 「Apps Script」。
  3. 貼上下方提供的 Apps Script 程式碼。
  4. 點擊 「部署」 ➔ 「新增部署」，類型選擇 「網頁應用程式」，存取權選 「所有人 (Anyone)」。
  5. 將獲得的網址貼回上方的輸入框即可完成連線！
// Google Apps Script 後端腳本 (義勝農場專用)
function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "義勝農場 Google Sheet API Ready" }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var data = JSON.parse(e.postData.contents);
  var action = data.action;

  if (action === "addField") {
    var sheet = ss.getSheetByName("FieldRecords");
    sheet.appendRow([data.vegType, data.plot, data.datePlant, data.dateFertilize, data.datePesticide1, data.datePesticide2, data.dateHarvest, data.dateHarvest2, data.harvestWeight1, data.harvestWeight2]);
  } else if (action === "addRaw") {
    var sheet = ss.getSheetByName("RawInbound");
    sheet.appendRow([data.lotNo, data.item, data.date, data.source, data.weight, data.zone, data.note]);
  } else if (action === "addShipment") {
    var sheet = ss.getSheetByName("Shipping");
    var itemsStr = JSON.stringify(data.items);
    sheet.appendRow([data.shipNo, data.customer, itemsStr, data.totalKg, data.date, data.temp, data.logistics]);
  }

  return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
    .setMimeType(ContentService.MimeType.JSON);
}
