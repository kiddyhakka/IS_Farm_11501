Google Sheets 雲端試算表串接設定

貼上您的 Google Apps Script 網頁應用程式 URL，系統將會將入庫、分裝與出貨即時儲存至您的 Google 試算表中。

步驟指南：如何建立免費 Google Sheets 後端 API

  1. 開啟全新的 Google 試算表，並於上方分頁建立三張工作表：RawInbound, Processing, Shipping。
  2. 點選上方選單 「擴充功能」 ➔ 「Apps Script」。
  3. 清空原有程式碼，貼上右下方點擊「複製程式碼」獲得的腳本。
  4. 點擊右上角 「部署」 ➔ 「新增部署」，選擇類型為 「網頁應用程式」。
  5. 「誰有存取權」請務必選為 「所有人 (Anyone)」，最後點擊「部署」並複製 URL 填入上方欄位即可！
// Google Apps Script 後端腳本
function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Google Sheet API Ready" }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var data = JSON.parse(e.postData.contents);
  var action = data.action;

  if (action === "addRaw") {
    var sheet = ss.getSheetByName("RawInbound");
    sheet.appendRow([data.lotNo, data.item, data.date, data.source, data.weight, data.zone, data.note]);
  } else if (action === "addProcess") {
    var sheet = ss.getSheetByName("Processing");
    sheet.appendRow([data.finishedLotNo, data.rawLotNo, data.date, data.rawWeightUsed, data.spec, data.packageCount, data.lossWeight, data.lossRate, data.operator]);
  } else if (action === "addShipment") {
    var sheet = ss.getSheetByName("Shipping");
    sheet.appendRow([data.shipNo, data.customer, data.finishedLotNo, data.count, data.date, data.temp, data.logistics]);
  }

  return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
    .setMimeType(ContentService.MimeType.JSON);
}
